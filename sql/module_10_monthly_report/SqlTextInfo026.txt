SELECT CASE Month_of_Year WHEN 1 THEN 'January' WHEN 2 THEN 'February' WHEN 3 THEN 'March' WHEN 4 THEN 'April' WHEN 5 THEN 'May' WHEN 6 THEN 'June' WHEN 7 THEN 'July' WHEN 8 THEN 'August' WHEN 9 THEN 'September' WHEN 10 THEN 'October' WHEN 11 THEN 'November' WHEN 12 THEN 'December' END (NAMED Month_of_Year ) , Avg_suspectCPUpct , CASE WHEN Avg_suspectCPUpct <= 25 THEN 'Healthy' WHEN Avg_suspectCPUpct BETWEEN 25.01 AND 44 THEN 'Degraded' ELSE 'Critical' END AS AvgSuspect_CPU , 'Suspect CPU' AS Suspect_CPU FROM Suspect_Cpu GROUP BY Month_of_Year , 2 , 3 , 4 ORDER BY Month_of_Year ASC ;
