SELECT CASE Month_of_Year WHEN 1 THEN 'January' WHEN 2 THEN 'February' WHEN 3 THEN 'March' WHEN 4 THEN 'April' WHEN 5 THEN 'May' WHEN 6 THEN 'June' WHEN 7 THEN 'July' WHEN 8 THEN 'August' WHEN 9 THEN 'September' WHEN 10 THEN 'October' WHEN 11 THEN 'November' WHEN 12 THEN 'December' END (NAMED Month_of_Year ) , AVG ( CPUUtilPct ) AS Avg_CpuUtil , CASE WHEN Avg_CpuUtil BETWEEN 0 AND 50 THEN 'Healthy' WHEN Avg_CpuUtil BETWEEN 50.01 AND 60 THEN 'Busy' WHEN Avg_CpuUtil BETWEEN 60.01 AND 80 THEN 'Very Busy' ELSE 'Critical' END AS CPU_Notation , 'CPU' AS CPU FROM CPU GROUP BY Month_of_Year ORDER BY Month_of_Year ;
